sat(A, B, C, Y) :-
	or(B, C, X),
	not(X, Z),
	and(A, Z, Y).
				\end{minted}
				\hrule
			\end{minipage}
			\caption{Prolog Code}

\begin{frame}[fragile]{QA-Prolog}{Esempio}
	\begin{minipage}{.48\linewidth}
		\begin{block}{Base di conoscenza}
			Il nemico del mio nemico è mio amico
		\end{block}
		\begin{figure}[h]
			\begin{minipage}{\linewidth}
				\hrule
				\begin{minted}{Prolog}
					
hates(alice, bob).
hates(bob, charlie).

enemies(P, Q) :- hates(P, Q).
enemies(P, Q) :- hates(Q, P).

friends(A, B) :-
	enemies(A, X),
	enemies(X, B),
	A \= B.
				\end{minted}
				\hrule
			\end{minipage}
			\caption{Enemy of my Enemy}
		\end{figure}
	\end{minipage}
	\hfill
	\pause
	\begin{minipage}{.48\linewidth}
		\begin{block}{Esecuzione}

				\scriptsize \texttt{QA-Prolog --qmasm-args="--postproc=opt" --query='friends(P1, P2).' friends.pl}
